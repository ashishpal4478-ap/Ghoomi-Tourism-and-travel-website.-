package com.ghoomi.common.config;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.Customizer;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.config.annotation.web.configurers.AbstractHttpConfigurer;
import org.springframework.security.config.http.SessionCreationPolicy;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;
import org.springframework.web.cors.CorsConfiguration;
import org.springframework.web.cors.CorsConfigurationSource;
import org.springframework.web.cors.UrlBasedCorsConfigurationSource;

import java.util.Arrays;
import java.util.List;

@Configuration
@EnableWebSecurity
public class SecurityConfig {

    @Value("${app.cors.allowed-origins:http://localhost:3000}")
    private String allowedOrigins;

    @Bean
    public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
        http
            .cors(Customizer.withDefaults())
            .csrf(AbstractHttpConfigurer::disable)
            .sessionManagement(session -> session.sessionCreationPolicy(SessionCreationPolicy.STATELESS))
            .authorizeHttpRequests(auth -> auth
                .requestMatchers("/api/v1/health").permitAll()
                .requestMatchers("/api/v1/auth/**").permitAll()
                .anyRequest().authenticated()
            );

        return http.build();
    }

    @Bean
    public CorsConfigurationSource corsConfigurationSource() {
        CorsConfiguration configuration = new CorsConfiguration();
        configuration.setAllowedOrigins(Arrays.stream(allowedOrigins.split(",")).map(String::trim).toList());
        configuration.setAllowedMethods(List.of("GET", "POST", "PUT", "PATCH", "DELETE", "OPTIONS"));
        configuration.setAllowedHeaders(List.of("Authorization", "Content-Type", "X-Requested-With", "Accept", "Origin", "Access-Control-Request-Method", "Access-Control-Request-Headers"));
        configuration.setAllowCredentials(true);
        configuration.setMaxAge(3600L);

        UrlBasedCorsConfigurationSource source = new UrlBasedCorsConfigurationSource();
        source.registerCorsConfiguration("/**", configuration);
        return source;
    }

    @Bean
    public PasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder(12);
    }
}

import com.fasterxml.jackson.annotation.JsonInclude;

import java.time.Instant;

@JsonInclude(JsonInclude.Include.NON_NULL)
public class ApiResponse<T> {

    private boolean success;
    private String message;
    private T data;
    private String timestamp;

    public ApiResponse() {
        this.timestamp = Instant.now().toString();
    }

    public ApiResponse(boolean success, String message, T data) {
        this.success = success;
        this.message = message;
        this.data = data;
        this.timestamp = Instant.now().toString();
    }

    public static <T> ApiResponse<T> ok(T data) {
        return new ApiResponse<>(true, "Operation successful", data);
    }

    public static <T> ApiResponse<T> ok(String message, T data) {
        return new ApiResponse<>(true, message, data);
    }

    public static <T> ApiResponse<T> success(String message) {

        this.data = data;

    }

    }

package com.ghoomi.common.exception;

import com.ghoomi.common.response.ApiResponse;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.FieldError;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

import java.util.stream.Collectors;

@RestControllerAdvice
public class GlobalExceptionHandler {

    private static final Logger log = LoggerFactory.getLogger(GlobalExceptionHandler.class);

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiResponse<Void>> handleValidationException(MethodArgumentNotValidException ex) {
        String errorMessage = ex.getBindingResult().getFieldErrors().stream()
                .map(FieldError::getDefaultMessage)
                .collect(Collectors.joining("; "));

        return ResponseEntity.status(HttpStatus.BAD_REQUEST)
                .body(ApiResponse.error(errorMessage.isEmpty() ? "Invalid request payload" : errorMessage));
    }

    @ExceptionHandler(IllegalArgumentException.class)
    public ResponseEntity<ApiResponse<Void>> handleIllegalArgumentException(IllegalArgumentException ex) {
        return ResponseEntity.status(HttpStatus.BAD_REQUEST)
                .body(ApiResponse.error(ex.getMessage()));
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<ApiResponse<Void>> handleGenericException(Exception ex) {
        log.error("Unhandled application exception", ex);
        return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR)
                .body(ApiResponse.error("An unexpected internal error occurred. Please try again later."));
    }
}

package com.ghoomi.modules.user;

import jakarta.persistence.*;
import java.time.LocalDateTime;
import java.time.ZoneOffset;
import java.util.UUID;

@Entity
@Table(name = "users")
public class User {

    @Id
    @Column(name = "id", columnDefinition = "CHAR(36)", nullable = false, updatable = false)
    private String id;

    @Column(name = "email", length = 254, nullable = false, unique = true)
    private String email;

    @Column(name = "password_hash", length = 255, nullable = false)
    private String passwordHash;

    @Column(name = "email_verified_at")
    private LocalDateTime emailVerifiedAt;

    @Enumerated(EnumType.STRING)
    @Column(name = "role", nullable = false, length = 191)
    private UserRole role = UserRole.USER;

    @Column(name = "is_active", nullable = false)
    private boolean active = true;

    @Column(name = "created_at", nullable = false, updatable = false)
    private LocalDateTime createdAt;

    @Column(name = "updated_at", nullable = false)
    private LocalDateTime updatedAt;

    // Compatibility fields: these are not columns in the users table.
    @Transient
    private String phoneE164;

    @Transient
    private String fullName;

    @Transient
    private boolean phoneVerified;

    @Transient
    private String accountType;

    @Transient
    private LocalDateTime lastLoginAt;

    protected User() {}

    public User(String email, String phoneE164, String passwordHash,
                String fullName, UserRole role,
                AccountStatus accountStatus, String accountType) {
        this.email = email;
        this.phoneE164 = phoneE164;
        this.passwordHash = passwordHash;
        this.fullName = fullName;
        this.role = role != null ? role : UserRole.USER;
        this.accountType = accountType;

        if (accountStatus != null) {
            this.active = accountStatus == AccountStatus.ACTIVE
                    || accountStatus == AccountStatus.PENDING_VERIFICATION;
        }
    }

    @PrePersist
    protected void onCreate() {
        if (id == null) {
            id = UUID.randomUUID().toString();
        }
        LocalDateTime now = LocalDateTime.now(ZoneOffset.UTC);
        if (createdAt == null) createdAt = now;
        if (updatedAt == null) updatedAt = now;
    }

    @PreUpdate
    protected void onUpdate() {
        updatedAt = LocalDateTime.now(ZoneOffset.UTC);
    }

    public String getId() { return id; }

    public String getEmail() { return email; }
    public void setEmail(String email) { this.email = email; }

    public String getPasswordHash() { return passwordHash; }
    public void setPasswordHash(String passwordHash) { this.passwordHash = passwordHash; }

    public UserRole getRole() { return role; }
    public void setRole(UserRole role) { this.role = role; }

    public boolean isActive() { return active; }
    public void setActive(boolean active) { this.active = active; }

    public boolean isEmailVerified() { return emailVerifiedAt != null; }

    public void setEmailVerified(boolean verified) {
        this.emailVerifiedAt = verified
                ? LocalDateTime.now(ZoneOffset.UTC) : null;
    }

    public LocalDateTime getEmailVerifiedAt() { return emailVerifiedAt; }
    public void setEmailVerifiedAt(LocalDateTime value) {
        this.emailVerifiedAt = value;
    }

    public String getPhoneE164() { return phoneE164; }
    public void setPhoneE164(String phoneE164) { this.phoneE164 = phoneE164; }

    public String getFullName() { return fullName; }
    public void setFullName(String fullName) { this.fullName = fullName; }

    public boolean isPhoneVerified() { return phoneVerified; }
    public void setPhoneVerified(boolean phoneVerified) {
        this.phoneVerified = phoneVerified;
    }

    public String getAccountType() { return accountType; }
    public void setAccountType(String accountType) { this.accountType = accountType; }

    public LocalDateTime getLastLoginAt() { return lastLoginAt; }
    public void setLastLoginAt(LocalDateTime lastLoginAt) {
        this.lastLoginAt = lastLoginAt;
    }

    public AccountStatus getAccountStatus() {
        return active ? AccountStatus.ACTIVE : AccountStatus.DEACTIVATED;
    }

    public void setAccountStatus(AccountStatus status) {
        this.active = status == AccountStatus.ACTIVE
                || status == AccountStatus.PENDING_VERIFICATION;
    }

    public LocalDateTime getCreatedAt() { return createdAt; }
    public LocalDateTime getUpdatedAt() { return updatedAt; }
}

    @Id

    @OneToOne(fetch = FetchType.LAZY)
    @MapsId

    @Column(name = "display_name", length = 150, nullable = false)

/**
 * JPA entity mapping the {@code otp_records} table (added in V2 migration).
 *
 * <p>
 * OTP values are stored as a BCrypt hash â€” never plaintext.
 * A record is consumed (single-use) on successful verification.
 */
@Entity
@Table(name = "otp_records")
public class OtpRecord {

    @Id
    @Column(name = "id", columnDefinition = "CHAR(36)", nullable = false, updatable = false)
    private String id;

    /** The email address this OTP was issued to. */
    @Column(name = "email", length = 254, nullable = false)
    private String email;

    /** BCrypt hash of the 6-digit OTP â€” never store plaintext. */
    @Column(name = "otp_hash", length = 72, nullable = false)
    private String otpHash;

    /**
     * Purpose discriminator â€” currently only PASSWORD_RESET is used in Partition 1.
     */
    @Enumerated(EnumType.STRING)
    @Column(name = "purpose", length = 30, nullable = false)
    private OtpPurpose purpose;

    /** UTC timestamp after which this OTP must be rejected. */
    @Column(name = "expires_at", nullable = false)
    private LocalDateTime expiresAt;

    /** Whether this record has already been used. */
    @Column(name = "used", nullable = false)
    private boolean used = false;

    /** Number of verification attempts made against this OTP. */
    @Column(name = "attempts", nullable = false)
    private byte attempts = 0;

    @Column(name = "created_at", nullable = false, updatable = false)
    private LocalDateTime createdAt;

    @PrePersist
    protected void onCreate() {
        if (id == null) {
            id = java.util.UUID.randomUUID().toString();

package com.ghoomi.modules.auth;

import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Modifying;
import org.springframework.data.jpa.repository.Query;
import org.springframework.stereotype.Repository;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDateTime;
import java.util.Optional;

/**
 * Spring Data JPA repository for {@link OtpRecord}.
 */
@Repository
public interface OtpRecordRepository extends JpaRepository<OtpRecord, String> {

    /**
     * Find the latest unused, unexpired OTP for a given email and purpose.
     * Ordered newest-first so that any re-issued OTP supersedes the previous one.
     */
    Optional<OtpRecord> findTopByEmailAndPurposeAndUsedFalseAndExpiresAtAfterOrderByCreatedAtDesc(
            String email, OtpPurpose purpose, LocalDateTime now);

    /**
     * Delete all OTP records for an email+purpose pair â€” called after a
     * successful reset to leave no dangling records.
     */
    @Modifying
    @Transactional
    @Query("DELETE FROM OtpRecord o WHERE o.email = :email AND o.purpose = :purpose")
    void deleteAllByEmailAndPurpose(String email, OtpPurpose purpose);
}

/**
 * Frontend-only integration contract. These are expected backend routes, NOT implemented APIs.
 * The backend must provide HTTP-only secure session cookies, CSRF protection, server-side
 * role validation and OTP issuing/verification. Never store a token in localStorage.
 */
export type ApiSuccess = { success: true; message?: string }
export type RegistrationInput = {
  name: string
  email: string
  phone: string
  password: string
  accountType: 'traveller' | 'partner_application'

  businessCategories?: string[]

  businessDetails?: {
    businessName: string
    businessType: string
    city: string
  }
}
export type LoginInput = { email: string; password: string; rememberMe: boolean }

const apiBase = process.env.NEXT_PUBLIC_AUTH_API_BASE_URL?.trim().replace(/\/$/, '')
export class AuthApiError extends Error {}

async function send<T extends ApiSuccess>(path: string, payload: object): Promise<T> {
  if (!apiBase) throw new AuthApiError('Authentication is not configured yet. Connect a real backend first.')
  let response: Response
  try {
    response = await fetch(`${apiBase}${path}`, {
      method: 'POST',
      credentials: 'include',
      cache: 'no-store',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(payload),
    })
  } catch {
    throw new AuthApiError('Authentication service cannot be reached. Please try again later.')
  }
  const result: unknown = await response.json().catch(() => null)
  const body = result && typeof result === 'object' ? result as Record<string, unknown> : {}
  if (!response.ok || body.success !== true) {
    const message = typeof body.message === 'string' && body.message.length < 250 ? body.message : 'The request could not be completed.'
    throw new AuthApiError(message)
  }
  return body as T
}

export const authApi = {
  login: (input: LoginInput) => send<ApiSuccess>('/auth/login', input),
  register: (input: RegistrationInput) => send<ApiSuccess>('/auth/register', input),
  forgotPassword: (email: string) => send<ApiSuccess & { retryAfterSeconds?: number }>('/auth/password/forgot', { email }),
  verifyOtp: (email: string, otp: string) => send<ApiSuccess>('/auth/password/verify-otp', { email, otp }),
  /** Requires a short-lived server-issued reset grant in an HttpOnly cookie. */
  resetPassword: (newPassword: string) => send<ApiSuccess>('/auth/password/reset', { newPassword }),
  async startGoogleAuth(): Promise<void> {
    const response = await send<ApiSuccess & { authorizationUrl?: string }>('/auth/oauth/google/start', { intent: 'sign_in' })
    if (!response.authorizationUrl) throw new AuthApiError('Google authentication is not configured.')
    let url: URL
    try { url = new URL(response.authorizationUrl) }
    catch { throw new AuthApiError('Authentication provider returned an invalid URL.') }
    if (url.protocol !== 'https:' && !(url.protocol === 'http:' && url.hostname === 'localhost')) {
      throw new AuthApiError('Authentication provider returned an unsafe URL.')
    }
    window.location.assign(url.toString())
  },
}

export function LoginForm({ active, onSwitch, className }: { active: boolean; onSwitch: () => void; className?: string }) {
  const [error, setError] = useState<string | null>(null)
  const [status, setStatus] = useState<string | null>(null)
  const [pending, setPending] = useState(false)
  const form = useFormState(FIELDS, validate, () => { setError(null); setStatus(null) })
  const s = (i: number) => staggerStyle(i, 1)
  const email = form.field('email')

  const handleSubmit = async (event: FormEvent<HTMLFormElement>) => {
    event.preventDefault()
    const element = event.currentTarget
    setStatus(null)
    if (!form.submit()) {
      requestAnimationFrame(() => element.querySelector<HTMLInputElement>('[aria-invalid="true"]')?.focus())

    setError(null)

      await authApi.login({ email: form.values.email.trim().toLowerCase(), password: form.values.password, rememberMe: Boolean(element.querySelector<HTMLInputElement>('[name="remember"]')?.checked) })
      setStatus('Sign in confirmed by the authentication service. Your dashboard will be available after it is built.')

    finally { setPending(false) }
  }
  const handleGoogle = async () => {
    setPending(true); setError(null)
    try { await authApi.startGoogleAuth() }
    catch (e) { setError(e instanceof Error ? e.message : 'Google authentication is unavailable.'); setPending(false) }
  }
  return (
    <FormShell active={active} className={className}>
      <div className="mb-8 flex flex-col gap-2">
        <h2 className="stagger-item font-display text-3xl font-bold text-foreground" style={s(0)}>Welcome back</h2>

const FIELDS = ['name', 'email', 'phone', 'password', 'confirm'] as const
type Values = Record<(typeof FIELDS)[number], string>
type AdminStep = 'category' | 'account' | 'business'
type SlideDirection = 'forward' | 'back'

function validate(values: Values) {
  const errors: Partial<Values> = {}

  if (!values.name.trim()) errors.name = 'Name is required'

  errors.email = emailError(values.email)

  const phone = values.phone.trim()
  if (!phone) errors.phone = 'Phone number is required'
  else if (!/^[6-9]\d{9}$/.test(phone)) {
    errors.phone = 'Enter a valid 10-digit phone number'
  }

  errors.password = passwordError(values.password)

  if (!values.confirm) errors.confirm = 'Confirm password is required'
  else if (values.confirm !== values.password) {
    errors.confirm = 'Passwords do not match'
  }

  return errors
}

function getBusinessFields(category: CategoryId) {
  if (category === 'hotels') {
    return { name: 'Hotel / Property Name', type: 'Property Type' }
  }

  if (category === 'restaurants') {
    return { name: 'Restaurant Name', type: 'Cuisine Type' }
  }

  return { name: 'Tour Company Name', type: 'Tour Service Type' }
}
